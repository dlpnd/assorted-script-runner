name DLpndMiscStatShot;

fn () {
    // Initialize and merge all arrays of NPCs and Creatures
    array aNPCs = GetLoadedTypeArray (42);
    array aCreature = GetLoadedTypeArray (43);
    Ar_Cat (aNPCs, aCreature);

    // Array to hold Body Parts to avoid init on a part that already were init'd
    array aBodyParts = [];
    for ([ref rActor] in aNPCs) {
        for (int i = 0; i < 15; i++) {
            string sBodyPartName = "Shot ${GetBodyPartName (GetBodyPartData (rActor), i)}";
            if (Sv_Length (sBodyPartName) > 5 && (Ar_Find (sBodyPartName, aBodyParts) == Ar_BadNumericIndex())) {
                Ar_Append (aBodyParts, sBodyPartName);
                InitExtraMiscStat (sBodyPartName);
            }
        }
    }
    Ar_Null (aBodyParts);

    SetOnHitEventHandler (fn () {
        ref rHitTarget = this();
        if (!rHitTarget.GetDead() && rHitTarget != PlayerRef && GetHitAttacker() == PlayerRef) {
            ModExtraMiscStat ("Shot ${GetBodyPartName (GetBodyPartData (rHitTarget.GetBaseObject()), rHitTarget.GetHitLocation())}", 1);
        }
    }, 1);
}
